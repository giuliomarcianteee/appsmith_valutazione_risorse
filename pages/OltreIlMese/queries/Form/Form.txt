SELECT 
    vr.IdMesi,
		vr.ModificatoDa,
		vr.DataCompilazione,
		vr.DataInserimento,
    vr.Accoglienza,
    vr.AccoglienzaSaluto,
    vr.AccoglienzaProporreAssistenza,
    vr.AccoglienzaIllustrarePosizionamentoNegozioEScontistiche,
    vr.AccoglienzaUscireDallaPenisola,
    vr.NoteAccoglienza,
    vr.AnalisiBisognoCliente,
    vr.AnalisiBisognoClienteDomandeAperte,
    vr.AnalisiBisognoClienteDomandeDiApprofondimento,
    vr.AnalisiBisognoClienteDomandeFinalizzateAllAcquisto,
    vr.AnalisiBisognoClienteRiformulareBisognoCliente,
    vr.NoteAnalisiBisognoCliente,
    d.Neg,
    d.Nome,
    d.IdDipendenti,
    d.OreSett,
    d.Livello,
    d.DataAssunzione,
    d.DataFineContratto,
    ln.Coordinatore
FROM valutazionerisorse_mesi AS vr
LEFT JOIN dipendenti AS d
ON vr.IdMesi = d.IdDipendenti
LEFT JOIN listanegozi AS ln
ON d.NEG = ln.NEG
WHERE d.IdDipendenti = {{appsmith.store.IdMesiSelezionato}};
