SELECT 
  d.*,
  ns.*
FROM dipendenti d
LEFT JOIN new_settimane ns ON ns.IdSettimane = d.IdDipendenti
WHERE d.NEG IN (
  SELECT NEG
  FROM listanegozi
  WHERE Coordinatore = '{{SelectCoordinatore.selectedOptionValue}}'
)
AND DATEDIFF(CURRENT_DATE, d.DataAssunzione) < 365

ORDER BY d.DataFineContratto ASC;
